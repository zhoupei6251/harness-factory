#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
quality-gate-runner.txt — 抖音口播脚本优化工作流 · 一键质量门运行器

功能：
  通过 Dify Workflow API 运行已导入的工作流，自动实现「score < 7 → 拼接 feedback 重跑」
  的质量门循环（最多 max_retries 次），无需在 Dify 画布中手动添加 Loop 节点。

使用方式：
  # 单次交互式运行（会提示输入参数）
  python quality-gate-runner.txt --api-base https://your-dify.com --api-key app-xxx --app-id app-xxx

  # 从 JSON 文件读取输入并运行
  python quality-gate-runner.txt --api-base https://your-dify.com --api-key app-xxx --app-id app-xxx --input inputs.json

  # 批量运行（CSV，每行一个 topic）
  python quality-gate-runner.txt --api-base https://your-dify.com --api-key app-xxx --app-id app-xxx --batch topics.csv

前置条件：
  1. 已将 dify-workflow-template.json 导入 Dify 并完成基础配置（换模型、绑变量）
  2. 在 Dify 工作流设置中开启「发布为 API」，获得 API Key 和 App ID
     （路径：工作流详情页 → API 访问 → 创建 API Key）

依赖：仅使用 Python 标准库（urllib / json / sys / os / time），无需 pip install。
"""

from __future__ import annotations

import json
import os
import sys
import time
import csv
import urllib.request
import urllib.error
import argparse
import io
import re

# ────────────────────── 配置常量 ──────────────────────

DEFAULT_API_BASE = ""          # 必须通过 --api-base 传入
DEFAULT_MAX_RETRIES = 2         # 质量门最大重跑次数（与文档一致）
DEFAULT_TIMEOUT = 300           # 单次 API 调用的总超时秒数（Node A 可能较慢）
BACKOFF_DELAYS = [3, 8]        # 网络失败时的指数退避间隔（秒）
QUALITY_GATE_THRESHOLD = 7      # score ≥ 7 放行


# ────────────────────── 核心逻辑 ──────────────────────

class QualityGateRunner:
    """Dify 工作流质量门运行器。封装 API 调用 + score 判定 + feedback 回灌循环。"""

    def __init__(self, api_base: str, api_key: str, app_id: str,
                 max_retries: int = DEFAULT_MAX_RETRIES,
                 timeout: int = DEFAULT_TIMEOUT):
        self.api_base = api_base.rstrip("/")
        self.api_key = api_key
        self.app_id = app_id
        self.max_retries = max_retries
        self.timeout = timeout
        self.run_count = 0

    # ─── API 调用层（含网络重试）───

    def _call_workflow(self, inputs: dict) -> dict:
        """
        调用 Dify Workflow Blocking Run API。
        自动处理网络超时 / 429 限流 / 5xx 服务端错误（指数退避重试）。
        返回解析后的 workflow outputs 字典。
        """
        url = f"{self.api_base}/v1/workflows/run"
        payload = {
            "inputs": inputs,
            "response_mode": "blocking",
            "user": "quality-gate-runner"
        }
        data = json.dumps(payload).encode("utf-8")

        headers = {
            "Authorization": f"Bearer {self.api_key}",
            "Content-Type": "application/json",
        }

        last_error = None
        for attempt, delay in enumerate([0] + BACKOFF_DELAYS):
            if attempt > 0:
                print(f"  ↻ 网络重试第 {attempt} 次（等待 {delay}s）…")
                time.sleep(delay)

            try:
                req = urllib.request.Request(url, data=data, headers=headers, method="POST")
                with urllib.request.urlopen(req, timeout=self.timeout) as resp:
                    body = json.loads(resp.read().decode("utf-8"))
                    return self._extract_outputs(body)
            except urllib.error.HTTPError as e:
                status = e.code
                err_body = ""
                try:
                    err_body = e.read().decode("utf-8", errors="replace")
                except Exception:
                    pass

                if status == 429:
                    last_error = f"API 限流 (HTTP 429)，请稍后重试"
                    continue
                elif 500 <= status < 600:
                    last_error = f"Dify 服务端错误 (HTTP {status}): {err_body[:200]}"
                    continue
                else:
                    raise RuntimeError(
                        f"[E004] Dify API 返回了意外状态（HTTP {status}）\n"
                        f"响应内容: {err_body[:200]}\n\n"
                        f"请按以下顺序排查（大概率是前两个之一）：\n"
                        f"  ① 打开 Dify 工作流设置页 → 确认「发布为 API」已开启且 API Key 没过期\n"
                        f"  ② 确认 App ID 复制正确（不是 Workflow ID）\n"
                        f"  ③ 在浏览器里访问 {self.api_base}，看是否能打开 Dify 页面\n"
                        f"  ④ 如果以上都没问题，可能是 Dify 服务端临时故障——等几秒重试"
                    )
            except urllib.error.URLError as e:
                last_error = f"网络连接失败: {e.reason}"
                continue
            except TimeoutError:
                last_error = f"请求超时 (> {self.timeout}s)，模型可能响应过慢"
                continue

        # 所有重试耗尽
        raise RuntimeError(
            f"[E004] Dify 服务连续 {len(BACKOFF_DELAYS)} 次都调不通，先暂停吧\n\n"
            f"最后一次报错: {last_error}\n\n"
            f"大概率是以下原因之一（按频率排序）：\n"
            f"  ① API Key 余额用完了 → 去 Dify 控制台充值或换一个 Key\n"
            f"  ② 网络不通（VPN/防火墙/代理问题）→ 在浏览器打开 Dify 地址试试能不能访问\n"
            f"  ③ 模型响应太慢（超过 {self.timeout} 秒）→ 换一个更快的模型或在 Dify 里把超时时间调大\n"
            f"  ④ Dify 服务端正在维护 → 等 1–2 分钟再试\n\n"
            f"修好之后重新运行这个命令就行，之前的输入不会丢。"
        )

    @staticmethod
    def _extract_outputs(body: dict) -> dict:
        """从 Dify API 响应中提取 workflow outputs。"""
        if "data" not in body:
            raise ValueError(f"API 响应格式异常（无 data 字段）: {json.dumps(body)[:500]}")

        data = body["data"]
        if "outputs" not in data:
            # workflow 可能以 error 形式返回
            if "error" in data:
                raise RuntimeError(f"[E004] 工作流执行报错: {data['error']}")
            raise ValueError(f"无法从 API 响应中提取 outputs: {json.dumps(data)[:500]}")

        return data["outputs"]

    # ─── feedback 构造器 ───

    @staticmethod
    def build_feedback(score: int, previous_output: dict) -> str:
        """根据上一轮 score 和输出，构造反馈文本回灌给 Node A。

        按 score 区间给出差异化改进指引，而非通用模板。
        """
        script_preview = previous_output.get("optimized_script", "")[:100]

        # 按分数段匹配最可能的扣分原因和具体改进建议
        if score >= 5:
            gap_hint = (
                "当前版本已经接近及格线（7分），但还差一口气。"
                "请重点检查以下 1–2 个维度做微调即可：\n"
                "- Hook 前三秒是否够「炸」？加一个反常识数据或冲突场景试试\n"
                "- 文案里是否全是正确废话？替换成「大多数人以为X，其实Y」这种有立场的表达\n"
                "- shot_plan 里有没有某个镜头的 visual_prompt 太空泛？补上具体物体名称\n\n"
                "注意：只微调弱项，已达标的维度不要动，避免越改越乱。"
            )
        elif score >= 3:
            gap_hint = (
                "当前版本有明显短板，需要针对性加强。请逐项检查：\n"
                "- 【Hook】开头是否有让人停下来的理由？（数据冲击/反常识结论/利益前置）\n"
                "- 【论证】每条论点是否有具体数字、案例或场景支撑？纯观点=没有说服力\n"
                "- 【结构】是否有清晰的「问题→拆解→方案/行动」推进节奏？\n"
                "- 【差异化】和同赛道其他视频比，你的独特角度是什么？如果没有，先定一个\n\n"
                "建议：先选最弱的 1–2 项重点改，不要全篇重写。"
            )
        else:
            gap_hint = (
                "当前版本与目标差距较大。请优先确认基础问题：\n"
                "- key_points 是否提供了足够多且具体的素材？（这是最常见的原因——输入太泛则输出一定泛）\n"
                "- topic 是否聚焦到可执行的具体方向？（「短视频运营」太宽，「为什么你的短视频没人看」才够窄）\n"
                "- 如果以上都没问题，尝试换一个更强的模型重新生成\n\n"
                "本轮反馈不要求你一次性改到位，先把骨架搭对再逐步精修。"
            )

        return (
            f"上一版得分：{score}/10（及格线 7 分）。\n\n"
            f"{gap_hint}\n\n"
            f"(上一版文案前100字参考: {script_preview}…)"
        )

    # ─── 主循环 ───

    def run(self, inputs: dict) -> dict:
        """
        执行带质量门的工作流运行。
        
        流程：
          1. 用原始 inputs 调用工作流
          2. 检查 score ≥ 7? → 直接返回结果
          3. score < 7 → 构造 feedback → 带 feedback 重跑（最多 max_retries 次）
          4. 仍 < 7 → 返回结果 + 质量预警标记
        
        返回字典包含所有产出字段 + 元数据（run_count, quality_warning 等）。
        """
        current_inputs = dict(inputs)  # 不污染原始参数
        current_inputs["feedback"] = ""  # 首次运行 feedback 为空
        last_output = {}
        final_score = 0

        for attempt in range(1, self.max_retries + 2):  # 1 + max_retries
            self.run_count = attempt
            label = "首跑" if attempt == 1 else f"重跑 #{attempt - 1}"
            print(f"\n{'='*50}")
            print(f"🎬 质量门运行 [{label}] ({attempt}/{self.max_retries + 1})")
            print(f"{'='*50}")

            output = self._call_workflow(current_inputs)
            last_output = output

            # 提取 score
            score = output.get("score")
            if score is None:
                print(f"⚠️ 未从输出中获取到 score 字段，跳过质量门判定")
                break

            final_score = score
            score_status = "✅ PASS" if score >= QUALITY_GATE_THRESHOLD else "❌ RETRY"
            print(f"\n📊 Score: {score}/10 [{score_status}]")

            if score >= QUALITY_GATE_THRESHOLD:
                print(f"🎉 质量门通过！（第 {attempt} 次尝试后达到 {score} 分）")
                break

            if attempt <= self.max_retries:
                # 还有重试机会 → 拼 feedback 并重跑
                feedback = self.build_feedback(score, output)
                current_inputs["feedback"] = feedback
                print(f"🔄 正在拼接 feedback 并重跑…")
                print(f"   Feedback 摘要: {feedback[:120]}…")
            else:
                # 重试次数耗尽
                print(f"\n⚠️ 质量预警：经过 {self.max_retries + 1} 次尝试，score 仍为 {score} (< 7)")
                print(f"   建议选项：① 手动优化后放行 ② 检查 key_points 质量 ③ 换更强的模型")

        # 组装最终结果
        result = {
            **last_output,
            "_meta": {
                "total_runs": self.run_count,
                "final_score": final_score,
                "quality_warning": final_score < QUALITY_GATE_THRESHOLD,
                "max_retries_used": self.run_count - 1,
            },
        }

        return result


# ────────────────────── 输入构建辅助 ──────────────────────

def build_inputs_from_args(args) -> dict:
    """从命令行参数或文件构建 Dify inputs 字典。"""
    if args.input:
        # 从 JSON 文件读
        with open(args.input, "r", encoding="utf-8") as f:
            return json.load(f)

    # 从命令行参数逐个构建
    inputs = {
        "raw_script": args.raw_script or "",
        "topic": args.topic or "",
        "key_points": args.key_points or "",
        "materials": args.materials or "",
        "content_type": args.content_type or "知识",
        "target_audience": args.target_audience or "",
        "video_duration": args.video_duration or 45,
        "account_persona": args.account_persona or "",
        "publish_time": "",
        "feedback": "",  # 由 runner 内部管理
    }
    return inputs


def interactive_input() -> dict:
    """交互式收集用户输入参数。"""
    print("\n" + "=" * 50)
    print("📝 抖音口播脚本优化 — 交互式输入模式")
    print("=" * 50)

    raw_script = input("初版口播文案（留空则走主题延展模式）: ").strip()
    topic = input("主题 [如: 为什么你的短视频没人看]: ").strip()
    kp_hint = "核心要点 3-5 条（用数字分隔，如: 1.完播率比点赞重要 2.前3秒决定生死）: "
    key_points = input(kp_hint).strip()

    content_type = input("内容类型 [知识/种草/剧情/观点] 默认=知识: ").strip() or "知识"
    target_audience = input("目标人群: ").strip() or ""
    video_duration_str = input("视频时长(秒) 默认=45: ").strip() or "45"

    try:
        video_duration = int(video_duration_str)
    except ValueError:
        video_duration = 45

    account_persona = input("账号人设（可选）: ").strip() or ""
    materials = input("素材/案例/数据（可选）: ").strip() or ""

    return {
        "raw_script": raw_script,
        "topic": topic,
        "key_points": key_points,
        "materials": materials,
        "content_type": content_type,
        "target_audience": target_audience,
        "video_duration": video_duration,
        "account_persona": account_persona,
        "publish_time": "",
        "feedback": "",
    }


def batch_run_csv(runner: QualityGateRunner, csv_path: str):
    """从 CSV 文件批量运行（每行一个任务）。"""
    results = []

    with open(csv_path, "r", encoding="utf-8-sig") as f:
        reader = csv.DictReader(f)
        rows = list(reader)

    print(f"\n📦 批量模式: 共 {len(rows)} 条任务\n")

    for i, row in enumerate(rows, 1):
        topic = row.get("topic", "").strip()
        key_points = (row.get("key_points") or row.get("核心要点") or "").strip()

        print(f"\n{'━'*60}")
        print(f"📋 批量任务 [{i}/{len(rows)}]: topic={topic}")
        print(f"{'━'*60}")

        inputs = {
            "raw_script": row.get("raw_script", ""),
            "topic": topic,
            "key_points": key_points,
            "materials": row.get("materials", ""),
            "content_type": row.get("content_type", "知识"),
            "target_audience": row.get("target_audience", ""),
            "video_duration": int(row.get("video_duration", 45)),
            "account_persona": row.get("account_persona", ""),
            "publish_time": "",
            "feedback": "",
        }

        try:
            result = runner.run(inputs)
            results.append({"row": i, "status": "ok", **result})
        except Exception as e:
            results.append({"row": i, "status": "error", "error": str(e)})
            print(f"❌ 任务 [{i}] 失败: {e}")

    # 写出汇总
    out_path = f"qg-results-{int(time.time())}.json"
    with open(out_path, "w", encoding="utf-8") as f:
        json.dump(results, f, ensure_ascii=False, indent=2)
    print(f"\n✅ 批量运行完成！结果已保存至: {out_path}")


# ────────────────────── CLI 入口 ──────────────────────

def main():
    parser = argparse.ArgumentParser(
        description="抖音口播脚本优化工作流 · 一键质量门运行器",
        formatter_class=argparse.RawDescriptionHelpFormatter,
        epilog="""
示例:
  # 交互式单次运行
  python quality-gate-runner.txt --api-base https://cloud.dify.ai --api-key app-xxxxx --app-id app-xxxxx

  # 从 JSON 文件运行
  python quality-gate-runner.txt --api-base https://cloud.dify.ai --api-key app-xxxxx --app-id app-xxxxx --input my-input.json

  # 批量运行 CSV
  python quality-gate-runner.txt --api-base https://cloud.dify.ai --api-key app-xxxxx --app-id app-xxxxx --batch topics.csv

  # 自定义重试次数
  python quality-gate-runner.txt ... --max-retries 3
""",
    )
    parser.add_argument("--api-base", required=True, help="Dify API 地址 (如 https://cloud.dify.ai)")
    parser.add_argument("--api-key", required=True, help="Dify 工作流 API Key（发布为 API 后创建）")
    parser.add_argument("--app-id", required=True, help="Dify 应用 ID")
    parser.add_argument("--input", help="输入 JSON 文件路径（跳过交互式输入）")
    parser.add_argument("--batch", help="批量运行 CSV 文件路径（需含 topic, key_points 列）")
    parser.add_argument("--max-retries", type=int, default=DEFAULT_MAX_RETRIES,
                        help=f"质量门最大重跑次数 (默认 {DEFAULT_MAX_RETRIES})")
    parser.add_argument("--timeout", type=int, default=DEFAULT_TIMEOUT,
                        help=f"单次 API 调用超时秒数 (默认 {DEFAULT_TIMEOUT})")
    # 单值参数（非交互模式时可用）
    parser.add_argument("--topic", default="", help="主题（非交互模式）")
    parser.add_argument("--key-points", default="", help="核心要点（非交互模式）")
    parser.add_argument("--raw-script", default="", help="初版口播文案（非交互模式）")
    parser.add_argument("--content-type", default="知识", help="内容类型 (默认 知识)")
    parser.add_argument("--target-audience", default="", help="目标人群")
    parser.add_argument("--video-duration", type=int, default=45, help="视频时长秒数")
    parser.add_argument("--account-persona", default="", help="账号人设")
    parser.add_argument("--materials", default="", help="素材/案例/数据")

    args = parser.parse_args()

    # 创建运行器
    runner = QualityGateRunner(
        api_base=args.api_base,
        api_key=args.api_key,
        app_id=args.app_id,
        max_retries=args.max_retries,
        timeout=args.timeout,
    )

    # 选择运行模式
    if args.batch:
        batch_run_csv(runner, args.batch)
        return

    # 构建 inputs
    if args.input or any([args.topic, args.key_points, args.raw_script]):
        inputs = build_inputs_from_args(args)
    else:
        inputs = interactive_input()

    # 输入预检（拦截 ~70% 运行前故障）
    errors = preflight_check(inputs)
    if errors:
        print("\n❌ 输入预检未通过：")
        for err in errors:
            print(f"   • {err}")
        print("\n请修正以上问题后重新运行。（对应错误码: E001/E002）")
        sys.exit(1)

    # 运行
    try:
        result = runner.run(inputs)

        # 打印关键结果
        print("\n" + "=" * 60)
        print("🎬 运行完成！以下是产出摘要:")
        print("=" * 60)

        meta = result.pop("_meta", {})
        score = result.get("score", "?")
        title_opts = result.get("title_options", [])
        script = result.get("optimized_script", "(空)")
        shot_plan = result.get("shot_plan", [])

        print(f"\n📊 最终 Score: {score}/10")
        if meta.get("quality_warning"):
            print(f"⚠️  质量预警标记: 是（{meta['total_runs']} 次尝试仍未达 7 分）")
        print(f"🔄 总调用次数: {meta.get('total_runs', '?')}")
        print(f"\n📝 优化后口播文案:\n   {script[:200]}{'…' if len(script) > 200 else ''}")

        print(f"\n🏷️  标题候选池 ({len(title_opts)} 个):")
        for j, t in enumerate(title_opts, 1):
            print(f"   {j}. {t}")

        print(f"\n🎬 Shot Plan ({len(shot_plan)} 个镜头槽):")
        for sp in shot_plan[:5]:
            print(f"   [{sp.get('beat','?')}] {sp.get('visual_prompt','?')[:60]}")
        if len(shot_plan) > 5:
            print(f"   …还有 {len(shot_plan)-5} 个槽")

        # 保存完整结果
        timestamp = int(time.time())
        out_file = f"douyin-result-{timestamp}.json"
        with open(out_file, "w", encoding="utf-8") as f:
            json.dump({**result, "_meta": meta}, f, ensure_ascii=False, indent=2)
        print(f"\n💾 完整结果已保存至: {out_file}")

    except KeyboardInterrupt:
        print("\n\n⚠️ 用户中断运行")
        sys.exit(130)
    except Exception as e:
        print(f"\n❌ 运行出错: {e}")
        sys.exit(1)


def preflight_check(inputs: dict) -> list[str]:
    """
    输入预检——在 LLM 调用之前拦截约 70% 的无效请求。
    返回错误列表（空 = 全部通过）。
    """
    errors = []

    # E001: key_points 或 topic/raw_script 至少一组有效
    has_raw = bool(inputs.get("raw_script", "").strip())
    has_topic = bool(inputs.get("topic", "").strip())
    has_kp = bool(inputs.get("key_points", "").strip())

    if not has_raw and (not has_topic or not has_kp):
        errors.append(
            "[E001] 输入不够——模型不知道要写什么方向。\n"
            "       你需要二选一：\n"
            "       ① 直接给一篇初稿（raw_script），我会帮你优化\n"
            "       ② 给一个主题（topic）+ 3–5 条核心要点（key_points），我来帮你从零写\n\n"
            "       ✅ 正确示例 key_points: '1.完播率比点赞重要 2.前3秒决定生死 3.算法赛马机制'\n"
            "       ❌ 错误示例 key_points: '短视频运营'（太泛了，这不是一条具体观点）"
        )
    elif has_kp:
        # 检查 key_points 数量（兼容中英文/数字/圈号/顿号等多种分隔符）
        kp_text = inputs["key_points"]
        segments = [s for s in re.split(
            r"[.\u3002\u3001\uff0c\uff1b;，。、]+|\d+[.\u3002]?|"
            r"[①②③④⑤⑥⑦⑧⑨⑩]|[一二三四五六七八九十]+[.\u3002]?",
            kp_text
        ) if s.strip()]
        count = len(segments)
        if count < 3:
            errors.append(
                f"[E001] key_points 只有约 {count} 条，需要至少 3 条才能写出有深度的文案。\n"
                f"       当前你填的: '{kp_text[:60]}{'…' if len(kp_text)>60 else ''}'\n\n"
                f"       💡 写法建议：每条要点应该是一个**可以被反驳的具体观点**，\n"
                f"          不是泛泛的话题词。用数字/顿号/换行分隔即可。\n"
                f"       ✅ 示例: '1.完播率比点赞重要 2.前3秒决定生死 3.算法赛马机制 4.同质化直接降权'"
            )

    # E002: video_duration 合法范围（自动修正，非致命，不阻塞运行）
    dur = inputs.get("video_duration")
    if isinstance(dur, (int, float)) and (dur < 15 or dur > 120):
        corrected = min(max(int(dur), 15), 120)
        print(f"  ⚠️ [E002] video_duration={dur} 超出推荐范围 [15, 120]，已自动修正为 {corrected} 秒。")
        inputs["video_duration"] = corrected

    return errors
